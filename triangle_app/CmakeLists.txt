cmake_minimum_required(VERSION 3.29)

project(VulkanTriangleApp)

include(FetchContent)
FetchContent_Declare(
  glfw
  GIT_REPOSITORY https://github.com/glfw/glfw.git
  GIT_TAG 3.4
)
FetchContent_MakeAvailable(glfw)
find_package(Vulkan 1.4.335 REQUIRED)

add_library(VulkanCppModule INTERFACE)
add_library(Vulkan::cppm ALIAS VulkanCppModule)
target_link_libraries(VulkanCppModule INTERFACE Vulkan::Vulkan)
target_compile_definitions(VulkanCppModule
  INTERFACE VULKAN_HPP_DISPATCH_LOADER_DYNAMIC=1 VULKAN_HPP_NO_STRUCT_CONSTRUCTORS=1
)

function (add_slang_shader_target TARGET)
  cmake_parse_arguments("SHADER" "" "APP_NAME" "SOURCES" ${ARGN})
  if(NOT SHADER_SOURCES)
    message(FATAL_ERROR "No shader source files provided for target ${TARGET}")
  endif()

  set(SHADERS_DIR ${CMAKE_CURRENT_BINARY_DIR}/${SHADER_APP_NAME}/shaders)
  set(ENTRY_POINTS -entry vertMain -entry fragMain)
  set(SHADER_OUTPUT ${SHADERS_DIR}/${TARGET}.spv)
  add_custom_command(
    OUTPUT  ${SHADER_OUTPUT}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${SHADERS_DIR}
    COMMAND ${SLANGC_EXECUTABLE} ${SHADER_SOURCES} -target spirv -profile spirv_1_4+spvRayQueryKHR -emit-spirv-directly -fvk-use-entrypoint-name ${ENTRY_POINTS} -o ${SHADER_OUTPUT}
    WORKING_DIRECTORY ${SHADERS_DIR}
    DEPENDS ${SHADERS_DIR} ${SHADER_SOURCES}
    COMMENT "Compiling slang shaders"
    VERBATIM
  )
  add_custom_target(${TARGET} DEPENDS ${SHADER_OUTPUT})
endfunction()

function (add_app APP_NAME)
  cmake_parse_arguments(APP "" "SHADER" "" ${ARGN})
  file(GLOB APP_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
  add_executable(${APP_NAME} ${APP_SOURCES})
  set_target_properties(${APP_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${APP_NAME}
  )
  set_target_properties(${APP_NAME} PROPERTIES CXX_STANDARD 20)
  target_link_libraries(${APP_NAME} PRIVATE Vulkan::cppm glfw)
  target_include_directories(${APP_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})

  target_compile_definitions(${APP_NAME} PRIVATE "VULKAN_HPP_HANDLE_ERROR_OUT_OF_DATE_AS_SUCCESS")

  if(WIN32)
    if(${CMAKE_GENERATOR} MATCHES "Visual Studio.*")
      set_target_properties(${APP_NAME} PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/${APP_NAME}")
    endif()
  endif()

  if (DEFINED APP_SHADER)
    set(APP_SHADER_SLANG_TARGET ${APP_NAME}_slang_shader)
    file(GLOB SHADER_SLANG_SOURCES ${APP_SHADER}.slang)
    if(SHADER_SLANG_SOURCES)
      add_slang_shader_target(${APP_SHADER_SLANG_TARGET} APP_NAME ${APP_NAME} SOURCES ${SHADER_SLANG_SOURCES})
      add_dependencies(${APP_NAME} ${APP_SHADER_SLANG_TARGET})
    endif()
  endif()
endfunction()

add_app(triangle_app
  SHADER shaders/triangle_shader
)
